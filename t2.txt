#!/usr/bin/env python3
"""
alcohol_alert_digital.py
Digital alcohol sensor -> buzzer + take USB cam photo -> send email with photo attachment.
"""

import time, os, ssl, smtplib
from email.message import EmailMessage

from gpiozero import Buzzer, DigitalInputDevice
try:
    import cv2
    HAVE_CV2 = True
except Exception:
    HAVE_CV2 = False

import requests  # used to get approximate location via ipinfo

# ---------- CONFIG ----------
# Hardware
DIGITAL_PIN = 4          # GPIO pin connected to module D0 (BCM numbering)
BUZZER_PIN = 17          # GPIO for buzzer (BCM)
SENSOR_ACTIVE_HIGH = True   # Set False if your module pulls the pin LOW when triggered

# Photo & file
PHOTO_PATH = "/home/pi/pothole_demo/images/alcohol_alert.jpg"
os.makedirs(os.path.dirname(PHOTO_PATH), exist_ok=True)

# Email (replace with real values)
SMTP_SERVER = "smtp.gmail.com"
SMTP_PORT = 465
EMAIL_USER = "youremail@gmail.com"        # -> replace
EMAIL_PASS = "your_app_password"          # -> replace with app password
AUTHORITY_EMAIL = "authority@example.com" # -> replace

# Behavior
CHECK_INTERVAL = 1.5      # seconds between checks
ALERT_COOLDOWN = 60       # seconds between repeated alerts

# Email text
EMAIL_SUBJECT = "ALERT: Alcohol detected on device"
EMAIL_BODY = "Alcohol detected on helmet device. See attached photo. Approx location/link included."

# ----------------------------

def capture_photo_cv2(path):
    try:
        cap = cv2.VideoCapture(0)
        if not cap.isOpened():
            print("cv2: cannot open camera")
            return False
        # a short warmup
        time.sleep(0.5)
        ret, frame = cap.read()
        cap.release()
        if not ret:
            print("cv2: capture failed")
            return False
        cv2.imwrite(path, frame)
        print("Saved photo (cv2) ->", path)
        return True
    except Exception as e:
        print("cv2 capture error:", e)
        return False

def capture_photo_fswebcam(path):
    cmd = f"fswebcam -r 1280x720 --no-banner {path}"
    rc = os.system(cmd)
    if rc == 0:
        print("Saved photo (fswebcam) ->", path)
        return True
    print("fswebcam failed (rc=%s)" % rc)
    return False

def get_approx_location_text():
    try:
        r = requests.get("https://ipinfo.io/json", timeout=4).json()
        loc = r.get("loc")
        city = r.get("city", "")
        region = r.get("region", "")
        country = r.get("country", "")
        if loc:
            lat, lon = loc.split(",")
            gmap = f"https://www.google.com/maps/search/?api=1&query={lat},{lon}"
            return f"{city} {region} {country}", gmap
    except Exception:
        pass
    return "Unknown", "N/A"

def send_email_with_attachment(to_addr, subject, body, attach_path=None):
    msg = EmailMessage()
    msg["From"] = EMAIL_USER
    msg["To"] = to_addr
    msg["Subject"] = subject
    msg.set_content(body)
    if attach_path and os.path.exists(attach_path):
        with open(attach_path, "rb") as f:
            data = f.read()
        import mimetypes
        ctype, _ = mimetypes.guess_type(attach_path)
        if ctype:
            maintype, subtype = ctype.split("/",1)
        else:
            maintype, subtype = "application", "octet-stream"
        msg.add_attachment(data, maintype=maintype, subtype=subtype, filename=os.path.basename(attach_path))

    try:
        context = ssl.create_default_context()
        with smtplib.SMTP_SSL(SMTP_SERVER, SMTP_PORT, context=context) as s:
            s.login(EMAIL_USER, EMAIL_PASS)
            s.send_message(msg)
        print("Email sent to", to_addr)
        return True
    except Exception as e:
        print("Email failed:", e)
        return False

def main():
    print("Starting digital alcohol monitor (D0) ...")
    buzzer = Buzzer(BUZZER_PIN)
    sensor = DigitalInputDevice(DIGITAL_PIN, pull_up=True)  # use pull_up True usually

    last_alert_time = 0

    try:
        while True:
            active = sensor.is_active
            # If your module is active-low set SENSOR_ACTIVE_HIGH=False meaning .is_active corresponds to not triggered
            triggered = active if SENSOR_ACTIVE_HIGH else (not active)
            print("Sensor active:", active, "=> triggered:", triggered)
            if triggered and (time.time() - last_alert_time) > ALERT_COOLDOWN:
                print("ALCOHOL TRIGGERED! -> buzzer + photo + email")
                # buzzer beep pattern
                for _ in range(5):
                    buzzer.on()
                    time.sleep(0.18)
                    buzzer.off()
                    time.sleep(0.12)

                # take photo
                ok = False
                if HAVE_CV2:
                    ok = capture_photo_cv2(PHOTO_PATH)
                if not ok:
                    ok = capture_photo_fswebcam(PHOTO_PATH)

                # location info
                city, gmap = get_approx_location_text()
                body = EMAIL_BODY + f"\nLocation: {city}\nMap: {gmap}\nTimestamp: {time.ctime()}\n"

                # send email
                attach = PHOTO_PATH if os.path.exists(PHOTO_PATH) else None
                send_email_with_attachment(AUTHORITY_EMAIL, EMAIL_SUBJECT, body, attach_path=attach)

                last_alert_time = time.time()
            time.sleep(CHECK_INTERVAL)
    except KeyboardInterrupt:
        print("Stopping monitor")
    finally:
        buzzer.off()

if _name_ == "_main_":
    main()
